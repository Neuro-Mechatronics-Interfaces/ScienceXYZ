cmake_minimum_required(VERSION 3.22)
project(exo_control VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(APPLE)
  set(CMAKE_INSTALL_RPATH "@loader_path")
elseif(UNIX AND NOT ANDROID)
  set(CMAKE_INSTALL_RPATH "$ORIGIN")
endif()
if(MSVC AND VCPKG_TARGET_TRIPLET MATCHES "windows-static$")
  # CMake CMP0091 defaults to /MD even with this vcpkg static CRT triplet.
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
option(BUILD_SHARED_LIBS "Build portable shared C ABI" ON)
option(EXO_BUILD_JNI "Build Java/Android JNI bridge" ${ANDROID})
if(EXO_BUILD_JNI AND NOT BUILD_SHARED_LIBS)
  message(FATAL_ERROR "JNI requires BUILD_SHARED_LIBS=ON; for a static C SDK disable EXO_BUILD_JNI")
endif()
include(CTest)
include(GNUInstallDirs)
find_package(Protobuf CONFIG QUIET)
if(NOT Protobuf_FOUND)
  find_package(Protobuf REQUIRED)
endif()
find_package(gRPC CONFIG REQUIRED)
find_package(cppzmq CONFIG REQUIRED)
find_package(Threads REQUIRED)

# Explicit HOST tools are mandatory when cross compiling; never execute target
# Android binaries as code generators. Their protobuf version must match target.
set(EXO_PROTOC "" CACHE FILEPATH "Host protoc executable")
set(EXO_GRPC_PLUGIN "" CACHE FILEPATH "Host grpc_cpp_plugin executable")
if(CMAKE_CROSSCOMPILING AND (NOT EXO_PROTOC OR NOT EXO_GRPC_PLUGIN))
  message(FATAL_ERROR "Cross build requires EXO_PROTOC and EXO_GRPC_PLUGIN host executables")
endif()
if(NOT EXO_PROTOC)
  set(EXO_PROTOC $<TARGET_FILE:protobuf::protoc>)
endif()
if(NOT EXO_GRPC_PLUGIN)
  set(EXO_GRPC_PLUGIN $<TARGET_FILE:gRPC::grpc_cpp_plugin>)
endif()
set(GEN "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${GEN}")
set(PROTOS "${CMAKE_CURRENT_SOURCE_DIR}/proto/gui_control.proto")
file(GLOB_RECURSE API_PROTOS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/proto/api/*.proto")
list(APPEND PROTOS ${API_PROTOS})
set(GENERATED)
foreach(PROTO IN LISTS PROTOS)
  file(RELATIVE_PATH REL "${CMAKE_CURRENT_SOURCE_DIR}/proto" "${PROTO}")
  string(REGEX REPLACE "\\.proto$" "" STEM "${REL}")
  add_custom_command(OUTPUT "${GEN}/${STEM}.pb.cc" "${GEN}/${STEM}.pb.h"
    COMMAND "${EXO_PROTOC}" --proto_path=${CMAKE_CURRENT_SOURCE_DIR}/proto
      --cpp_out=${GEN} "${PROTO}"
    DEPENDS ${PROTOS} VERBATIM)
  list(APPEND GENERATED "${GEN}/${STEM}.pb.cc" "${GEN}/${STEM}.pb.h")
endforeach()
add_custom_command(OUTPUT "${GEN}/api/synapse.grpc.pb.cc" "${GEN}/api/synapse.grpc.pb.h"
  COMMAND "${EXO_PROTOC}" --proto_path=${CMAKE_CURRENT_SOURCE_DIR}/proto
    --grpc_out=${GEN} --plugin=protoc-gen-grpc=${EXO_GRPC_PLUGIN}
    "${CMAKE_CURRENT_SOURCE_DIR}/proto/api/synapse.proto"
  DEPENDS ${PROTOS} VERBATIM)
add_library(exo_wire STATIC ${GENERATED} "${GEN}/api/synapse.grpc.pb.cc")
target_include_directories(exo_wire PUBLIC $<BUILD_INTERFACE:${GEN}> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}/exo_control/wire>)
target_link_libraries(exo_wire PUBLIC protobuf::libprotobuf gRPC::grpc++)
add_library(exo_core STATIC src/controller.cpp src/synapse_transport.cpp)
target_include_directories(exo_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(exo_core PUBLIC exo_wire PRIVATE cppzmq Threads::Threads)
add_library(exo_control src/c_api.cpp)
target_include_directories(exo_control PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(exo_control PRIVATE exo_core)
target_compile_definitions(exo_control PRIVATE EXO_BUILD)
if(ANDROID)
  # Do not interpose our statically linked protobuf/gRPC symbols into another
  # native SDK loaded by the same Android process. Keep the C ABI visible.
  target_link_options(exo_control PRIVATE "-Wl,--exclude-libs,ALL"
    "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
  set_target_properties(exo_control PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
endif()
if(NOT BUILD_SHARED_LIBS)
  target_compile_definitions(exo_control PUBLIC EXO_STATIC)
endif()
if(NOT ANDROID)
  set_target_properties(exo_control PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION 1)
endif()
add_library(exo::control ALIAS exo_control)
add_library(exo::core ALIAS exo_core)
set_target_properties(exo_control PROPERTIES EXPORT_NAME control)
set_target_properties(exo_core PROPERTIES EXPORT_NAME core)
set_target_properties(exo_wire PROPERTIES EXPORT_NAME wire)
if(EXO_BUILD_JNI)
  if(NOT ANDROID)
    find_package(JNI REQUIRED)
  endif()
  add_library(exo_jni SHARED src/jni.cpp)
  target_include_directories(exo_jni PRIVATE ${JNI_INCLUDE_DIRS})
  target_link_libraries(exo_jni PRIVATE exo_control)
  if(ANDROID)
    target_link_options(exo_jni PRIVATE "-Wl,--exclude-libs,ALL" "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
    set_target_properties(exo_jni PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
  endif()
  install(TARGETS exo_jni LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
  if(NOT CMAKE_CROSSCOMPILING)
    find_package(Java 11 REQUIRED COMPONENTS Development Runtime)
    set(JAVA_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/android/src/main/java/org/sciencexyz/exo/ExoClient.java")
    add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/exo-client.jar"
      COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_CURRENT_BINARY_DIR}/java-main"
      COMMAND "${Java_JAVAC_EXECUTABLE}" --release 8 -d "${CMAKE_CURRENT_BINARY_DIR}/java-main" "${JAVA_SOURCE}"
      COMMAND "${Java_JAR_EXECUTABLE}" cf "${CMAKE_CURRENT_BINARY_DIR}/exo-client.jar" -C "${CMAKE_CURRENT_BINARY_DIR}/java-main" .
      DEPENDS "${JAVA_SOURCE}" VERBATIM)
    add_custom_target(exo_java ALL DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/exo-client.jar")
    install(FILES "${CMAKE_CURRENT_BINARY_DIR}/exo-client.jar" DESTINATION share/exo_control/java)
  endif()
endif()
if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
  add_executable(exo_tests tests/controller_test.cpp)
  target_link_libraries(exo_tests PRIVATE exo_core exo_control)
  add_test(NAME exo-controller COMMAND exo_tests)
  set_tests_properties(exo-controller PROPERTIES TIMEOUT 20)
  add_executable(exo_loopback_tests tests/loopback_test.cpp)
  target_link_libraries(exo_loopback_tests PRIVATE exo_core exo_control cppzmq Threads::Threads)
  add_test(NAME exo-loopback COMMAND exo_loopback_tests)
  set_tests_properties(exo-loopback PROPERTIES TIMEOUT 20)
  if(EXO_BUILD_JNI)
    add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/java-test/JniSmoke.class"
      COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_CURRENT_BINARY_DIR}/java-test"
      COMMAND "${Java_JAVAC_EXECUTABLE}" --release 8 -d "${CMAKE_CURRENT_BINARY_DIR}/java-test"
        "${JAVA_SOURCE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/JniSmoke.java"
      DEPENDS "${JAVA_SOURCE}" tests/JniSmoke.java VERBATIM)
    add_custom_target(exo_java_tests ALL DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/java-test/JniSmoke.class")
    add_test(NAME exo-jni COMMAND "${Java_JAVA_EXECUTABLE}"
      "-Djava.library.path=$<TARGET_FILE_DIR:exo_jni>" -cp "${CMAKE_CURRENT_BINARY_DIR}/java-test" JniSmoke)
    set_tests_properties(exo-jni PROPERTIES TIMEOUT 20)
  endif()
endif()
install(TARGETS exo_control exo_core exo_wire EXPORT ExoControlTargets
  RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(DIRECTORY "${GEN}/" DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/exo_control/wire FILES_MATCHING PATTERN "*.h")
install(DIRECTORY android/src/main/java/ DESTINATION share/exo_control/java)
install(FILES README.md DESTINATION share/exo_control)
install(DIRECTORY docs/ DESTINATION share/exo_control/docs)
install(DIRECTORY proto/ DESTINATION share/exo_control/proto)
set(EXO_DEPENDENCY_SHARE "${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/share" CACHE PATH "Dependency notice directory (vcpkg share)")
if(IS_DIRECTORY "${EXO_DEPENDENCY_SHARE}")
  file(GLOB NOTICES "${EXO_DEPENDENCY_SHARE}/*/copyright")
  foreach(NOTICE IN LISTS NOTICES)
    get_filename_component(PACKAGE_DIR "${NOTICE}" DIRECTORY)
    get_filename_component(PACKAGE_NAME "${PACKAGE_DIR}" NAME)
    install(FILES "${NOTICE}" DESTINATION share/exo_control/notices RENAME "${PACKAGE_NAME}.txt")
  endforeach()
endif()
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/ExoControlConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/ExoControlConfig.cmake" INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/ExoControl)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/ExoControlConfigVersion.cmake" VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(EXPORT ExoControlTargets NAMESPACE exo:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/ExoControl)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/ExoControlConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/ExoControlConfigVersion.cmake" DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/ExoControl)
set(CPACK_PACKAGE_NAME exo-control)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_GENERATOR ZIP)
include(CPack)
